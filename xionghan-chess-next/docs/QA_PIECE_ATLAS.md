# perf_atlas.py —— 只读复用现有 API，不改源码
import sys, time
sys.path.insert(0, "src")
from PySide6.QtWidgets import QApplication
from PySide6.QtGui import QPixmap
from xionghan_chess.core.game import Game
from xionghan_chess.core.profiles import PROFILES
from xionghan_chess.core.model import Color
from xionghan_chess.desktop.ui.board_widget import BoardWidget

size = int(sys.argv[1]); style = sys.argv[2]
app = QApplication([])
board = BoardWidget()
board.set_game(Game(PROFILES["desktop_complete"]), Color.RED)
board.set_appearance("classic", "none", style)
board.resize(size, size); board.show()
pm = QPixmap(board.size() * board.devicePixelRatioF())
board.render(pm)                       # 预热
t = time.perf_counter()
for _ in range(60):
    board.render(pm)
print(f"{size}px {style}: {(time.perf_counter()-t)/60*1000:.2f} ms/frame")

## 10. 待裁定与已知问题

### T-1 ⚠ D-3 的「200% DPI 下盘径 ≥ 108 物理 px」经测算不可达

以 `board_geometry.py:29-31` 的公式实测推算（1080p 全屏 200% → 逻辑 960×540）：

| 量 | 推算值 |
|---|---|
| 棋盘可用高度 ≈ 540 − 菜单栏 ≈22 − banner ≈26 − 上下留白 2×13 | ≈ **466 逻辑 px** |
| cell = 466 / (13 − 1 + 1.35) | ≈ **34.9 逻辑 px** |
| 盘径 = 0.78 × cell | ≈ **27.2 逻辑 px** = **54.4 物理 px** |

即使按 100% DPI 算，1080p 全屏盘径也只有 ≈58 物理 px。**108 物理 px 需要盘径占屏高 ≥10%、约 4K 屏才能达到**，与「1080p 200%」互相矛盾。

**建议**（等主理人裁定，本文不擅自改基线）：
- 方案①：阈值改为 **≥ 54 物理 px**（1024×768 之类小屏再议）；
- 方案②：改为相对判据「盘径 / 逻辑 cell = 0.78 ± 1%，且 ≥ 200% 下理论值（0.78 × cell）的 95%」——与 D-2 合并，避免拍一个绝对像素数；
- 方案③：把「≥108 物理 px」移到 **4K / 200%** 场景另立一条。

在裁定前，D-3 按「记录实测值、判定填待裁定」执行。
